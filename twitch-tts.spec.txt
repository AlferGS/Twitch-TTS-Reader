datas = []

# ---------- 2. Данные (без весов моделей) ----------
for pkg in ['qfluentwidgets', 'xtts_api_server', 'num2words', 'anyascii',
            'inflect', 'gruut', 'gruut_lang_en']:
    try:

try:

# ---------- 4. Метаданные ----------

a = Analysis(

    binaries=binaries,

    hiddenimports=hiddenimports,
    hookspath=[],
    hooksconfig={},
    runtime_hooks=['rthook_torchscript.py'],
    excludes=[
    ],
    noarchive=False,

# ---------- 5. Хирургически вырезаем тяжёлые данные неиспользуемых языков ----------
# Удаляем ТОЛЬКО файлы данных (словари/веса), код пакетов остаётся —
# поэтому импорты не ломаются. Ошибка будет только если реально
# попытаетесь озвучить JP/DE/ES/FR.
def _drop_data(entry):
    name, src, typecode = entry
    low = src.replace('\\', '/').lower()
    if low.endswith(('.pth', '.pt')):                    # веса моделей (model.pth)
        return True
    if 'sudachidict' in low or low.endswith('.dic'):     # словарь JP, ~207 МБ
        return True
    if 'unidic' in low:                                  # словарь JP
        return True
    if 'gruut_lang_de' in low or 'gruut_lang_es' in low or 'gruut_lang_fr' in low:
        return True
    return False

_removed = [e for e in a.datas if _drop_data(e)]
a.datas = [e for e in a.datas if not _drop_data(e)]
_size = sum(os.path.getsize(e[1]) for e in _removed if os.path.exists(e[1]))
print(f"[Spec] Вырезано тяжёлых файлов: {len(_removed)}, {_size/1024**3:.2f} GB")

pyz = PYZ(a.pure, a.zipped_data)

exe = EXE(
    pyz, a.scripts, [],
    exclude_binaries=True,
    name='TwitchTTSReader',
    debug=False,
    strip=False,
    upx=False,
    console=True,          # ← временно True, чтобы видеть трейсбеки в консоли
    target_arch=None,
)

coll = COLLECT(

    strip=False,

)
